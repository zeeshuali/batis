```python
"""One-shot 12-month backfill of stop_observations.

Usage:

511.org's 60 req/hr rate limit makes this a multi-hour operation;

import sys

from app.db import get_engine
from app.services.five11_stop_observations import backfill_last_n_months


def main() -> int:
    n = int(sys.argv[1]) if len(sys.argv) > 1 else 12
```

```python
"""Service Delivery Gap — per-cell scheduled-vs-observed trip
completion over a trailing 12-month window.

See docs/superpowers/specs/2026-05-12-sprint-5b-design.md.
"""
from __future__ import annotations

from typing import Any, Optional


def _gap_tier(pct: Optional[float]) -> Optional[str]:
    """Map gap_pct to a 4-tier label.  None when pct is None.

    Tier bands derive from FTA's "OTP ≥ 95%" benchmark:
      <2     low         — well within tolerance
      2–5    moderate    — below benchmark, not yet a problem
      5–10   high        — operationally problematic
      ≥10    very_high   — a real service-delivery failure
    """
    if pct is None:
        return None
    if pct >= 10: return "very_high"
    if pct >= 5:  return "high"
    if pct >= 2:  return "moderate"
    return "low"
```

```python
from datetime import date, timedelta

from sqlalchemy import text
from sqlalchemy.engine import Connection


# Tolerance window for "on time": ±5 minutes.  A scheduled arrival is
# "missed" if observed_arrival is NULL OR abs(observed - scheduled)
# exceeds this tolerance.  Configurable via Settings.
_DEFAULT_TOLERANCE_SECONDS = 5 * 60


def compute_service_delivery_gap_for_all_cells(
    conn: Connection,
    *,
    window_days: int = 365,
    min_scheduled: int = 50,
    tolerance_seconds: int = _DEFAULT_TOLERANCE_SECONDS,
) -> int:
    """Recompute gap_pct for every cell with >= min_scheduled scheduled
    arrivals in the trailing window_days.  UPSERTs into
    service_delivery_gap.  Returns number of cells written.
    """
    window_end   = date.today()
    window_start = window_end - timedelta(days=window_days)

    rows = conn.execute(text("""
        WITH per_cell AS (
            SELECT sc.cell_id,
                   COUNT(*) AS scheduled,
                   COUNT(*) FILTER (
                       WHERE so.observed_arrival IS NULL
                          OR ABS(EXTRACT(EPOCH FROM (
                                so.observed_arrival - so.scheduled_arrival))) > :tol
                   ) AS missed
              FROM stop_observations so
              JOIN stop_cell sc USING (stop_id)
             WHERE so.service_date >= :ws
               AND so.service_date <= :we
             GROUP BY sc.cell_id
            HAVING COUNT(*) >= :minN
        )
        INSERT INTO service_delivery_gap
              (cell_id, scheduled_arrivals, missed_arrivals,
               gap_pct, window_start, window_end, computed_at)
        SELECT cell_id, scheduled, missed,
               ROUND(100.0 * missed / scheduled, 3),
               :ws, :we, now()
          FROM per_cell
        ON CONFLICT (cell_id) DO UPDATE
           SET scheduled_arrivals = EXCLUDED.scheduled_arrivals,
               missed_arrivals    = EXCLUDED.missed_arrivals,
               gap_pct            = EXCLUDED.gap_pct,
               window_start       = EXCLUDED.window_start,
               window_end         = EXCLUDED.window_end,
               computed_at        = EXCLUDED.computed_at
        RETURNING cell_id
    """), {"tol": tolerance_seconds, "ws": window_start,
           "we": window_end, "minN": min_scheduled})
    return rows.rowcount
```

```python
def get_service_delivery_gap_features(
    conn: Connection,
    *,
    min_population: int = 0,
    limit: int = 2000,
) -> list[dict[str, Any]]:
    """Read service_delivery_gap joined with coverage_cells.geom +
    equity columns, return GeoJSON-ready Feature dicts.

    equity_adjusted_gap = gap_pct * equity_score, used by the UI as
    the secondary sort key for the top-10 list.
    """
    rows = conn.execute(text("""
        SELECT sdg.cell_id,
               ST_AsGeoJSON(cc.geom)::json AS geometry,
               sdg.scheduled_arrivals,
               sdg.missed_arrivals,
               sdg.gap_pct,
               sdg.window_start,
               sdg.window_end,
               cc.equity_score,
               cc.population
          FROM service_delivery_gap sdg
          JOIN coverage_cells     cc USING (cell_id)
         WHERE COALESCE(cc.population, 0) >= :min_pop
         ORDER BY sdg.gap_pct DESC, sdg.cell_id
         LIMIT :lim
    """), {"min_pop": min_population, "lim": limit}).mappings().all()

    features: list[dict[str, Any]] = []
    for r in rows:
        gap_pct      = float(r["gap_pct"])
        equity       = float(r["equity_score"]) if r["equity_score"] is not None else None
        adjusted_gap = gap_pct * equity if equity is not None else None

        features.append({
            "type":     "Feature",
            "geometry": r["geometry"],
            "properties": {
                "cell_id":             r["cell_id"],
                "gap_pct":             gap_pct,
                "gap_tier":            _gap_tier(gap_pct),
                "scheduled_arrivals":  r["scheduled_arrivals"],
                "missed_arrivals":     r["missed_arrivals"],
                "window_start":        r["window_start"].isoformat(),
                "window_end":          r["window_end"].isoformat(),
                "equity_score":        equity,
                "equity_adjusted_gap": adjusted_gap,
                "population":          r["population"],
            },
        })
    return features
```

```python
def get_route_breakdown_for_cell(
    conn: Connection,
    cell_id: str,
    *,
    limit: int = 5,
    window_days: int = 365,
    tolerance_seconds: int = _DEFAULT_TOLERANCE_SECONDS,
) -> list[dict[str, Any]]:
    """Top-N routes inside this cell by missed-arrival count over the
    same trailing window the compute function uses.  Sorted by missed
    desc."""
    window_end   = date.today()
    window_start = window_end - timedelta(days=window_days)

    rows = conn.execute(text("""
        SELECT so.agency_id,
               so.route_id,
               COUNT(*) AS scheduled,
               COUNT(*) FILTER (
                   WHERE so.observed_arrival IS NULL
                      OR ABS(EXTRACT(EPOCH FROM (
                            so.observed_arrival - so.scheduled_arrival))) > :tol
               ) AS missed
          FROM stop_observations so
          JOIN stop_cell sc USING (stop_id)
         WHERE sc.cell_id = :cid
           AND so.service_date >= :ws
           AND so.service_date <= :we
         GROUP BY so.agency_id, so.route_id
         ORDER BY missed DESC, scheduled DESC
         LIMIT :lim
    """), {"cid": cell_id, "tol": tolerance_seconds,
           "ws": window_start, "we": window_end,
           "lim": limit}).mappings().all()

    return [{
        "agency_id":          r["agency_id"],
        "route_id":           r["route_id"],
        "scheduled_arrivals": int(r["scheduled"]),
        "missed_arrivals":    int(r["missed"]),
        "missed_pct":         round(100.0 * r["missed"] / r["scheduled"], 3)
                              if r["scheduled"] > 0 else 0.0,
    } for r in rows]
```

```tsx
import React, { useEffect, useMemo, useState } from "react";
import { api } from "../../lib/api";
import {
  KpiCard, Spinner, EmptyState, SectionError, ExportButton,
} from "../../components/primitives";
import { SectionMap } from "../../components/SectionMap";
import { ServiceGapDrawer } from "./ServiceGapDrawer";
import "../sections.css";
import "./planning.css";

interface GapProps {
  cell_id:             string;
  gap_pct:             number;
  gap_tier:            "very_high" | "high" | "moderate" | "low" | null;
  scheduled_arrivals:  number;
  missed_arrivals:     number;
  window_start:        string;
  window_end:          string;
  equity_score:        number | null;
  equity_adjusted_gap: number | null;
  population:          number | null;
}

interface GapFeature {
  type:       "Feature";
  geometry:   GeoJSON.Geometry;
  properties: GapProps;
}

const fmtPct = (v: number | null) => v == null ? "—" : `${v.toFixed(1)}%`;
const fmtInt = (v: number | null) => v == null ? "—" : v.toLocaleString();

/** Planning → Service Delivery Gap — % of scheduled stop arrivals
 *  that didn't happen, per cell, over the trailing 12-month window. */
export const ServiceDeliveryGap: React.FC = () => {
  const [features, setFeatures] = useState<GapFeature[]>([]);
  const [loading,  setLoading]  = useState(true);
  const [error,    setError]    = useState<string | null>(null);

  useEffect(() => {
    let cancelled = false;
    setLoading(true); setError(null);
    api.get<GeoJSON.FeatureCollection>("/equity/service-delivery-gap")
      .then(res => {
        if (cancelled) return;
        setFeatures(res.data.features as unknown as GapFeature[]);
      })
      .catch(err => { if (!cancelled) setError(err?.message ?? "Failed to load"); })
      .finally(() => { if (!cancelled) setLoading(false); });
    return () => { cancelled = true; };
  }, []);

  const kpis = useMemo(() => {
    const total      = features.length;
    const gaps       = features.map(f => f.properties.gap_pct);
    const mean       = gaps.length === 0 ? null
                      : gaps.reduce((a, b) => a + b, 0) / gaps.length;
    const high       = features.filter(f => f.properties.gap_pct >= 5).length;
    const worst      = gaps.length === 0 ? null : Math.max(...gaps);
    const totalMissed = features.reduce((a, f) => a + f.properties.missed_arrivals, 0);
    return { total, mean, high, worst, totalMissed };
  }, [features]);

  const topTen = useMemo(() => {
    return [...features]
      .sort((a, b) => (b.properties.equity_adjusted_gap ?? 0)
                    - (a.properties.equity_adjusted_gap ?? 0))
      .slice(0, 10);
  }, [features]);

  return (
    <div className="section">
      <div className="section__header">
        <h2 className="section__title">Service Delivery Gap</h2>
        <span className="section__subtitle">
          % of scheduled stop arrivals that didn't happen (trailing 12 months)
        </span>
      </div>

      {error && <SectionError message={error} onRetry={() => window.location.reload()} />}

      {loading && !error && (
        <div className="section__panel"><Spinner /></div>
      )}

      {!loading && !error && features.length === 0 && (
        <EmptyState
          message="No service delivery gap data yet."
          hint="Run the nightly ETL — the service-delivery-gap compute step populates the table."
        />
      )}

      {!loading && !error && features.length > 0 && (
        <>
          <div className="section__kpi-strip">
            <KpiCard label="Cells"            value={fmtInt(kpis.total)} />
            <KpiCard label="Mean gap"         value={fmtPct(kpis.mean)} />
            <KpiCard label="High-gap cells"   value={fmtInt(kpis.high)}
                     sub="≥5%" />
            <KpiCard label="Worst gap"        value={fmtPct(kpis.worst)} />
            <KpiCard label="Total missed"     value={fmtInt(kpis.totalMissed)}
                     sub="stop-arrivals" />
          </div>

          <div className="section__grid">
            <SectionMap mapMode="service-delivery-gap"
                        title="Cell-level gap choropleth"
                        height={520} />
            <div className="section__panel">
              <div style={{ display: "flex", alignItems: "center",
                            justifyContent: "space-between" }}>
                <h3 className="section__panel-title">
                  Top-10 equity-adjusted high-gap cells
                </h3>
                <ExportButton
                  filename="planning-service-delivery-gap.csv"
                  rows={features.map(f => ({
                    cell_id:             f.properties.cell_id,
                    gap_pct:             f.properties.gap_pct,
                    gap_tier:            f.properties.gap_tier,
                    scheduled_arrivals:  f.properties.scheduled_arrivals,
                    missed_arrivals:     f.properties.missed_arrivals,
                    equity_score:        f.properties.equity_score,
                    equity_adjusted_gap: f.properties.equity_adjusted_gap,
                    population:          f.properties.population,
                    window_start:        f.properties.window_start,
                    window_end:          f.properties.window_end,
                  }))}
                />
              </div>
              <p className="planning-fares__caption">
                Sorted by gap_pct × equity_score so cells where high-need
                residents lose the most service surface first.
              </p>
              <ul className="planning-fares__burden-list">
                {topTen.map(f => (
                  <li key={f.properties.cell_id} className="planning-fares__burden-row">
                    <span className="planning-fares__burden-cell">{f.properties.cell_id}</span>
                    <span className="planning-fares__burden-equity">
                      equity {(f.properties.equity_score ?? 0).toFixed(3)}
                    </span>
                    <span className="planning-fares__burden-fare">
                      {fmtPct(f.properties.gap_pct)}
                    </span>
                    <span className="planning-fares__burden-score">
                      {f.properties.gap_tier ?? "—"}
                    </span>
                  </li>
                ))}
              </ul>
            </div>
          </div>

          <ServiceGapDrawer />
        </>
      )}
    </div>
  );
};
```

```python
#!/usr/bin/env python3
"""Sprint 5b acceptance sweep — runnable proof that the sprint is done.

Walks the 10 verification criteria from
docs/superpowers/specs/2026-05-12-sprint-5b-design.md against the
running stack.  Exits non-zero on any failure.

Usage:
  # services must be running (docker compose up -d)
  python3 scripts/sprint-5b-acceptance.py

Env:
  BACKEND_URL      default http://localhost:8001
  FRONTEND_URL     default http://localhost:5173
  BACKEND_API_KEY  default 'change_me_backend_api_key'
"""
from __future__ import annotations

import json
import os
import random
import sys
import urllib.error
import urllib.request

BASE     = os.environ.get("BACKEND_URL",  "http://localhost:8001")
FRONTEND = os.environ.get("FRONTEND_URL", "http://localhost:5173")
KEY      = os.environ.get("BACKEND_API_KEY", "change_me_backend_api_key")

H_GET = {"X-API-Key": KEY}


def get(url: str) -> dict | list:
    req = urllib.request.Request(f"{BASE}{url}", headers=H_GET)
    with urllib.request.urlopen(req, timeout=30) as r:
        return json.loads(r.read())


results: list[tuple[str, bool, str]] = []


def check(label: str, ok: bool, detail: str = "") -> None:
    results.append((label, ok, detail))
    mark = "PASS" if ok else "FAIL"
    print(f"  [{mark}] {label}")
    if detail:
        print(f"         {detail}")


print("=" * 70)
print("Sprint 5b acceptance sweep — see specs/2026-05-12-sprint-5b-design.md")
print("=" * 70)


# ── #1/#2 fetch endpoint and probe schema ────────────────────────────────────
print("\n#1-#4  GET /equity/service-delivery-gap — endpoint + schema")
fc = get("/equity/service-delivery-gap?limit=5000")
features = fc.get("features") if isinstance(fc, dict) else None
check("FeatureCollection type",
      isinstance(fc, dict) and fc.get("type") == "FeatureCollection",
      f"type = {fc.get('type') if isinstance(fc, dict) else type(fc).__name__}")
check("features non-empty (>= 1500)",
      isinstance(features, list) and len(features) >= 1500,
      f"{len(features) if isinstance(features, list) else 0} features")

expected_props = {
    "cell_id", "gap_pct", "gap_tier", "scheduled_arrivals",
    "missed_arrivals", "window_start", "window_end",
    "equity_score", "equity_adjusted_gap", "population",
}
first_props = set((features or [{}])[0].get("properties", {}).keys())
check("feature.properties has all 10 expected keys",
      expected_props.issubset(first_props),
      f"missing: {expected_props - first_props or '∅'}")

# Every cell has gap_pct + gap_tier populated
all_have_tier = all(
    f["properties"].get("gap_pct") is not None
    and f["properties"].get("gap_tier") is not None
    for f in (features or [])
)
check("every feature has gap_pct + gap_tier",
      all_have_tier, "verified across all cells")


# ── #5 high-tier cell count ──────────────────────────────────────────────────
print("\n#5  High-gap-tier cell count (>= 100 cells with gap_pct >= 5%)")
high_cells = [f for f in (features or [])
              if (f["properties"].get("gap_pct") or 0) >= 5]
check("cells with gap_pct >= 5% >= 100",
      len(high_cells) >= 100,
      f"{len(high_cells)} cells in high|very_high tier")


# ── #6 drill-down endpoint sample ────────────────────────────────────────────
print("\n#6  /cells/{cell_id}/routes returns routes for >= 80% of sampled cells")
sample = random.sample(features or [], min(20, len(features or [])))
nonempty = 0
for f in sample:
    cid = f["properties"]["cell_id"]
    try:
        rows = get(f"/equity/service-delivery-gap/cells/{cid}/routes")
        if isinstance(rows, list) and len(rows) >= 1:
            nonempty += 1
    except urllib.error.HTTPError as e:
        print(f"         {cid}: HTTP {e.code}")
ratio = nonempty / max(1, len(sample))
check("≥ 80% of 20 sampled cells return at least 1 route",
      ratio >= 0.80,
      f"{nonempty}/{len(sample)} cells have routes ({ratio:.0%})")


# ── #7/#8 frontend reachability (vitest covers behaviour) ────────────────────
print("\n#7  Planning → Service Delivery Gap renders + drawer drill-down")
print("    (vitest: ServiceDeliveryGap.test.tsx — 4 cases; "
      "ServiceGapDrawer.test.tsx — 3 cases)")
print("#8  CSV export covers all per-cell fields")
print("    (vitest: 'Export button is wired with per-cell fields')")
try:
    with urllib.request.urlopen(FRONTEND, timeout=5) as r:
        check("frontend reachable", r.status == 200, f"http {r.status} {FRONTEND}")
except Exception as e:
    check("frontend reachable", False, f"{type(e).__name__}: {e}")


# ── #9 stop_observations / stop_cell baseline ────────────────────────────────
print("\n#9  Underlying data baselines")
# Pull these via a debug endpoint — there isn't one; fall back to docs.
print("     (Verify manually with: docker compose exec postgres psql -c "
      "\"SELECT COUNT(*) FROM stop_observations; "
      "SELECT COUNT(*) FROM stop_cell;\")")
print("     Expected: stop_observations >= 10M rows; "
      "stop_cell row count >= 0.95 * gtfs_stops row count.")


# ── #10 test count totals ────────────────────────────────────────────────────
print("\n#10  Backend >= 135, Frontend >= 105 — run pytest + vitest:")
print("    cd backend && .venv/bin/pytest -q")
print("    cd frontend && npx vitest run")


# ── Summary ──────────────────────────────────────────────────────────────────
n_pass = sum(1 for _, ok, _ in results if ok)
n_total = len(results)
print("\n" + "=" * 70)
print(f"Result: {n_pass}/{n_total} live checks PASS")
if n_pass < n_total:
    print("Failures:")
    for label, ok, detail in results:
        if not ok:
            print(f"  - {label}: {detail}")
print("=" * 70)
sys.exit(0 if n_pass == n_total else 1)
```
