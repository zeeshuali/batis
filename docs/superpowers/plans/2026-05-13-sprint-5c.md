> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship Sprint 5c per `docs/superpowers/specs/2026-05-13-sprint-5c-design.md` — a Planning → ADA Gap sub-page ranking Bay Area transit stops by an ADA-accessibility gap score (`supply_penalty × ambulatory_pop_400m`), with per-stop click-drill showing routes + contributing tracts. Cell rollup heatmap as secondary toggle.

**Architecture:** Three new loaders (Census TIGERweb tract geometry; GTFS Pathways; ACS B18105 ambulatory pop) feed a single SQL pass (`compute_stop_ada_gap_for_all_stops`) that writes the `stop_ada_gap` fact table per ETL run, with `cell_ada_gap` as a derived materialized view. Three new FastAPI endpoints expose stops FeatureCollection, cell FC, and per-stop breakdown JSON. A new React Planning sub-page renders the stops as colored/sized circles on the map plus a top-10 table, with the cell rollup as a legend-toggle heatmap.

**Tech Stack:** Python 3.12 + FastAPI + SQLAlchemy `text()` + PostgreSQL/PostGIS + React 18 + TypeScript + Vite + Mapbox GL JS 3.x + Recharts (via existing primitives). Backend in Docker on port 8001, frontend on 5173.

```sql
-- Sprint 5c — ADA Accessibility Gap Tracker
-- See docs/superpowers/specs/2026-05-13-sprint-5c-design.md

-- Census tract geometry for the 9-county Bay Area (loaded from
-- TIGERweb GeoJSON REST). Required for areal interpolation that maps
-- tract-level ambulatory_pop into per-stop catchment buffers.
CREATE TABLE IF NOT EXISTS census_tracts (
  tract_geoid text PRIMARY KEY,
  county_fips text NOT NULL,
  geom        geometry(MultiPolygon, 4326),
  computed_at timestamptz DEFAULT now()
);
CREATE INDEX IF NOT EXISTS census_tracts_geom_idx
  ON census_tracts USING GIST (geom);

-- GTFS Pathways static data (per-station internal graph)
CREATE TABLE IF NOT EXISTS gtfs_pathways (
  agency_id        text NOT NULL,
  pathway_id       text NOT NULL,
  from_stop_id     text NOT NULL,
  to_stop_id       text NOT NULL,
  pathway_mode     integer,
  is_bidirectional boolean,
  length_m         numeric,
  traversal_time   integer,
  stair_count      integer,
  max_slope        numeric,
  PRIMARY KEY (agency_id, pathway_id)
);
CREATE INDEX IF NOT EXISTS gtfs_pathways_from_idx
  ON gtfs_pathways (agency_id, from_stop_id);
CREATE INDEX IF NOT EXISTS gtfs_pathways_to_idx
  ON gtfs_pathways (agency_id, to_stop_id);

CREATE TABLE IF NOT EXISTS gtfs_levels (
  agency_id   text NOT NULL,
  level_id    text NOT NULL,
  level_index numeric,
  level_name  text,
  PRIMARY KEY (agency_id, level_id)
);

-- Tract-level ambulatory-difficulty population from ACS B18105
CREATE TABLE IF NOT EXISTS census_ambulatory (
  tract_geoid    text PRIMARY KEY,
  ambulatory_pop integer NOT NULL,
  total_pop      integer,
  computed_at    timestamptz DEFAULT now()
);

-- Per-stop ADA gap score (fact table)
CREATE TABLE IF NOT EXISTS stop_ada_gap (
  stop_id              text PRIMARY KEY,
  agency_id            text,
  supply_penalty       numeric(3,2),
  ambulatory_pop_400m  integer,
  gap_score            numeric(10,2),
  gap_tier             text,
  pathways_override    boolean,
  catchment_radius_m   integer,
  computed_at          timestamptz DEFAULT now()
);
CREATE INDEX IF NOT EXISTS stop_ada_gap_tier_idx
  ON stop_ada_gap (gap_tier);
CREATE INDEX IF NOT EXISTS stop_ada_gap_score_idx
  ON stop_ada_gap (gap_score DESC);

-- Derived per-cell rollup
CREATE MATERIALIZED VIEW IF NOT EXISTS cell_ada_gap AS
  SELECT cc.cell_id,
         COUNT(sag.stop_id)                                AS stop_count,
         MAX(sag.gap_score)                                AS max_gap_score,
         AVG(sag.gap_score)                                AS mean_gap_score,
         (ARRAY_AGG(sag.gap_tier ORDER BY
            CASE sag.gap_tier
              WHEN 'very_high' THEN 4
              WHEN 'high'      THEN 3
              WHEN 'moderate'  THEN 2
              WHEN 'low'       THEN 1
              ELSE 0
            END DESC NULLS LAST))[1]                       AS worst_tier
    FROM coverage_cells cc
    LEFT JOIN stop_cell    sc  ON sc.cell_id  = cc.cell_id
    LEFT JOIN stop_ada_gap sag ON sag.stop_id = sc.stop_id
   GROUP BY cc.cell_id;
CREATE INDEX IF NOT EXISTS cell_ada_gap_score_idx
  ON cell_ada_gap (max_gap_score DESC NULLS LAST);
```

Note: `stop_ada_gap` has no FK to `gtfs_stops(stop_id)` because that column lacks a unique constraint (same precedent as Sprint 5a's `fare_burden` and Sprint 5b's `service_delivery_gap`). `worst_tier` uses `ARRAY_AGG` + ordering because PostgreSQL doesn't support `MAX(... ORDER BY)` directly.

```python
"""Census TIGERweb tract geometry loader for the 9-county Bay Area.

See docs/superpowers/specs/2026-05-13-sprint-5c-design.md.

The Census Bureau's TIGERweb REST service serves authoritative tract
polygons as GeoJSON, sidestepping the need for geopandas / pyshp.
"""
from __future__ import annotations

import json
import logging
from typing import Any

import certifi
import requests
from sqlalchemy import text
from sqlalchemy.engine import Connection


logger = logging.getLogger(__name__)


# 9-county Bay Area FIPS — county code only (state code 06 is California).
# Matches _COUNTIES in equity_loader.py.
_COUNTY_FIPS = [
    ("001", "Alameda"),
    ("013", "Contra Costa"),
    ("041", "Marin"),
    ("055", "Napa"),
    ("075", "San Francisco"),
    ("081", "San Mateo"),
    ("085", "Santa Clara"),
    ("095", "Solano"),
    ("097", "Sonoma"),
]

_TIGERWEB_BASE = (
    "https://tigerweb.geo.census.gov/arcgis/rest/services/TIGERweb/"
    "Tracts_Blocks/MapServer/3/query"
)


def _fetch_county(county_fips: str) -> list[dict[str, Any]]:
    """Fetch tract GeoJSON features for one Bay Area county."""
    url = (
        f"{_TIGERWEB_BASE}?where=STATE%3D06+AND+COUNTY%3D{county_fips}"
        "&outFields=GEOID,STATE,COUNTY,TRACT"
        "&outSR=4326&f=geojson"
    )
    resp = requests.get(url, timeout=60, verify=certifi.where())
    resp.raise_for_status()
    data = resp.json()
    return list(data.get("features", []))


def load_bay_area_tracts(conn: Connection) -> int:
    """Fetch and upsert Bay Area tract geometries.

    Idempotent: ON CONFLICT (tract_geoid) DO UPDATE.  Promotes Polygon
    geometry to MultiPolygon via ST_Multi for column-type compatibility.

    Returns rowcount written across all 9 counties.
    """
    total = 0
    for county_fips, county_name in _COUNTY_FIPS:
        try:
            features = _fetch_county(county_fips)
        except Exception as e:  # noqa: BLE001
            logger.warning("[tracts] %s (%s) fetch failed: %s",
                           county_name, county_fips, e)
            continue

        for feat in features:
            props = feat.get("properties") or {}
            geoid = props.get("GEOID")
            geom = feat.get("geometry")
            if not geoid or not geom:
                continue
            conn.execute(text("""
                INSERT INTO census_tracts (tract_geoid, county_fips, geom, computed_at)
                VALUES (
                    :geoid, :county,
                    ST_Multi(ST_GeomFromGeoJSON(:geom_json)),
                    now()
                )
                ON CONFLICT (tract_geoid) DO UPDATE
                  SET county_fips = EXCLUDED.county_fips,
                      geom        = EXCLUDED.geom,
                      computed_at = EXCLUDED.computed_at
            """), {"geoid": geoid, "county": county_fips,
                   "geom_json": json.dumps(geom)})
            total += 1
        logger.info("[tracts] %s (%s): %d features", county_name,
                    county_fips, len(features))
    return total
```

```python
"""GTFS Pathways loader for monitored Bay Area agencies.

See docs/superpowers/specs/2026-05-13-sprint-5c-design.md.

Most agencies don't publish pathways.txt — the loader silently skips
those.  Idempotent per agency_id.
"""
from __future__ import annotations

import csv
import io
import logging
import zipfile
from typing import Optional

import certifi
import requests
from sqlalchemy import text
from sqlalchemy.engine import Connection

from app.core.config import get_settings


logger = logging.getLogger(__name__)


def _maybe_int(s: str | None) -> Optional[int]:
    if s is None or s == "":
        return None
    try:
        return int(s)
    except (TypeError, ValueError):
        return None


def _maybe_float(s: str | None) -> Optional[float]:
    if s is None or s == "":
        return None
    try:
        return float(s)
    except (TypeError, ValueError):
        return None


def _maybe_bool(s: str | None) -> Optional[bool]:
    if s is None or s == "":
        return None
    return s.strip() == "1"


def load_pathways_from_zip_bytes(
    conn: Connection, agency_id: str, zip_bytes: bytes
) -> int:
    """Parse pathways.txt and levels.txt out of a GTFS zip byte blob,
    upsert into gtfs_pathways and gtfs_levels.  Returns combined row
    count (pathways + levels).  Idempotent: deletes existing rows for
    agency_id before re-insert.
    """
    pathways_rows = 0
    levels_rows = 0

    # Idempotency: clear agency-scoped rows up front.
    conn.execute(text("DELETE FROM gtfs_pathways WHERE agency_id = :a"),
                 {"a": agency_id})
    conn.execute(text("DELETE FROM gtfs_levels   WHERE agency_id = :a"),
                 {"a": agency_id})

    with zipfile.ZipFile(io.BytesIO(zip_bytes)) as zf:
        if "pathways.txt" in zf.namelist():
            with zf.open("pathways.txt") as f:
                reader = csv.DictReader(
                    io.TextIOWrapper(f, encoding="utf-8-sig")
                )
                for row in reader:
                    conn.execute(text("""
                        INSERT INTO gtfs_pathways
                          (agency_id, pathway_id, from_stop_id, to_stop_id,
                           pathway_mode, is_bidirectional, length_m,
                           traversal_time, stair_count, max_slope)
                        VALUES (:a, :pid, :f, :t, :m, :b, :l, :tt, :sc, :ms)
                    """), {
                        "a":   agency_id,
                        "pid": row.get("pathway_id"),
                        "f":   row.get("from_stop_id"),
                        "t":   row.get("to_stop_id"),
                        "m":   _maybe_int(row.get("pathway_mode")),
                        "b":   _maybe_bool(row.get("is_bidirectional")),
                        "l":   _maybe_float(row.get("length")),
                        "tt":  _maybe_int(row.get("traversal_time")),
                        "sc":  _maybe_int(row.get("stair_count")),
                        "ms":  _maybe_float(row.get("max_slope")),
                    })
                    pathways_rows += 1

        if "levels.txt" in zf.namelist():
            with zf.open("levels.txt") as f:
                reader = csv.DictReader(
                    io.TextIOWrapper(f, encoding="utf-8-sig")
                )
                for row in reader:
                    conn.execute(text("""
                        INSERT INTO gtfs_levels
                          (agency_id, level_id, level_index, level_name)
                        VALUES (:a, :lid, :idx, :name)
                    """), {
                        "a":    agency_id,
                        "lid":  row.get("level_id"),
                        "idx":  _maybe_float(row.get("level_index")),
                        "name": row.get("level_name"),
                    })
                    levels_rows += 1

    if pathways_rows or levels_rows:
        logger.info("[pathways] %s: %d pathways + %d levels",
                    agency_id, pathways_rows, levels_rows)
    return pathways_rows + levels_rows


def load_all_pathways(conn: Connection) -> dict[str, int]:
    """Iterate `SELECT agency_id FROM agencies WHERE monitored = true`
    (or fall back to settings.five11_agencies if the agencies table
    is empty / missing) and load pathways for each.

    Returns {agency_id: rowcount} for agencies that contributed any
    rows; agencies whose GTFS zip lacks pathways.txt are not included
    in the result map.
    """
    s = get_settings()
    base = f"{s.five11_base_url}/transit/datafeeds"

    # Pull list of monitored agencies
    try:
        rows = conn.execute(text(
            "SELECT agency_id FROM agencies WHERE monitored IS TRUE "
            "ORDER BY agency_id"
        )).all()
        agencies = [r[0] for r in rows]
    except Exception:  # noqa: BLE001
        agencies = list(getattr(s, "five11_agencies", []) or [])

    result: dict[str, int] = {}
    for agency_id in agencies:
        try:
            url = f"{base}?operator_id={agency_id}&api_key={s.five11_api_key}"
            resp = requests.get(url, timeout=120, verify=certifi.where())
            resp.raise_for_status()
            n = load_pathways_from_zip_bytes(conn, agency_id, resp.content)
            if n > 0:
                result[agency_id] = n
```

```python
"""ACS B18105 (Sex by Age by Ambulatory Difficulty) loader for the
9-county Bay Area.  Sums ambulatory-difficulty bins across age and
sex to produce one ambulatory_pop value per tract.
"""
from __future__ import annotations

import logging
from typing import Any

import certifi
import requests
from sqlalchemy import text
from sqlalchemy.engine import Connection

from app.core.config import get_settings


logger = logging.getLogger(__name__)


_COUNTY_FIPS = [
    "001", "013", "041", "055", "075", "081", "085", "095", "097",
]

# "With ambulatory difficulty" subtotal cells across age × sex bins.
# Source: B18105 ACS table.
_AMBULATORY_BINS = [
    "B18105_004E", "B18105_007E", "B18105_010E",
    "B18105_013E", "B18105_016E", "B18105_019E",
    "B18105_023E", "B18105_026E", "B18105_029E",
    "B18105_032E", "B18105_035E", "B18105_038E",
]
_TOTAL_POP = "B18105_001E"


def _safe_int(s: str) -> int:
    """Cast ACS string to int, treating negative sentinels as 0."""
    try:
        v = int(s)
    except (TypeError, ValueError):
        return 0
    return v if v >= 0 else 0


def _fetch_county_tracts(county_fips: str) -> list[list[str]]:
    s = get_settings()
    var_list = ",".join([_TOTAL_POP] + _AMBULATORY_BINS)
    url = (
        f"https://api.census.gov/data/2022/acs/acs5"
        f"?get={var_list}"
        f"&for=tract:*"
        f"&in=state:06+county:{county_fips}"
        f"&key={s.census_api_key}"
    )
    resp = requests.get(url, timeout=60, verify=certifi.where())
    resp.raise_for_status()
    return resp.json()


def load_b18105(conn: Connection) -> int:
    """Fetch B18105 for every Bay Area tract, sum ambulatory-difficulty
    bins per tract, upsert into census_ambulatory.  Returns rowcount.
    """
    s = get_settings()
    if not s.census_api_key:
        logger.warning("[ambulatory] CENSUS_API_KEY missing; skipping load")
        return 0

    total_rows = 0
    for county in _COUNTY_FIPS:
        try:
            data = _fetch_county_tracts(county)
        except Exception as e:  # noqa: BLE001
            logger.warning("[ambulatory] county %s fetch failed: %s",
                           county, e)
            continue

        if not data or len(data) < 2:
            continue

        header = data[0]
        idx_total = header.index(_TOTAL_POP)
        idx_amb   = [header.index(b) for b in _AMBULATORY_BINS]
        idx_state = header.index("state")
        idx_cnty  = header.index("county")
        idx_tract = header.index("tract")

        for row in data[1:]:
            geoid = f"{row[idx_state]}{row[idx_cnty]}{row[idx_tract]}"
            total = _safe_int(row[idx_total])
            amb   = sum(_safe_int(row[i]) for i in idx_amb)
            conn.execute(text("""
                INSERT INTO census_ambulatory
                  (tract_geoid, ambulatory_pop, total_pop, computed_at)
                VALUES (:g, :a, :t, now())
                ON CONFLICT (tract_geoid) DO UPDATE
                  SET ambulatory_pop = EXCLUDED.ambulatory_pop,
                      total_pop      = EXCLUDED.total_pop,
                      computed_at    = EXCLUDED.computed_at
            """), {"g": geoid, "a": amb, "t": total})
            total_rows += 1
        logger.info("[ambulatory] county %s: %d tracts", county, len(data) - 1)
    return total_rows
```

```python
"""ADA Accessibility Gap — per-stop scoring of supply_penalty
(wheelchair_boarding + Pathways override) × demand (areal-
interpolated ACS B18105 ambulatory pop in 400m catchment).

See docs/superpowers/specs/2026-05-13-sprint-5c-design.md.
"""
from __future__ import annotations

import logging
from typing import Any, Optional

from sqlalchemy import text
from sqlalchemy.engine import Connection


logger = logging.getLogger(__name__)


def _supply_penalty(
    wheelchair_boarding: Optional[int],
    has_accessible_pathway: Optional[bool],
    location_type: Optional[int],
) -> float:
    """Map wheelchair_boarding + Pathways-override to penalty in [0, 1].

      wheelchair_boarding = 1 (accessible)        → 0.0
      wheelchair_boarding = 0 / None (unknown)    → 0.5
      wheelchair_boarding = 2 (not accessible)    → 1.0

    Pathways override (rail stations only — location_type in {1, 2}):
      has_accessible_pathway is False
        → 1.0 (regardless of wheelchair_boarding)

    has_accessible_pathway = None means "no pathways data for this
    stop" — do not override; fall through to wheelchair_boarding.
    """
    if location_type in (1, 2) and has_accessible_pathway is False:
        return 1.0
    if wheelchair_boarding == 1:
        return 0.0
    if wheelchair_boarding == 2:
        return 1.0
    return 0.5


def _gap_tier(score: float, *, p90: float, p75: float, p50: float) -> str:
    """Percentile-banded tier label.

    Stops with score == 0 always return 'low' (already accessible —
    no investment need regardless of demand).
    """
    if score <= 0:
        return "low"
    if score >= p90:
        return "very_high"
    if score >= p75:
        return "high"
    if score >= p50:
        return "moderate"
    return "low"
```

```python
def compute_stop_ada_gap_for_all_stops(
    conn: Connection,
    *,
    catchment_radius_m: Optional[int] = None,
) -> int:
    """Recompute per-stop ADA gap scores in one SQL pass.

    CTE pipeline:
      supply       — wheelchair_boarding + Pathways override (rail only)
      demand       — areal-interpolated B18105 into a buffer of
                     `catchment_radius_m` meters around each stop
      combined     — supply × demand → gap_score
      percentiles  — PERCENTILE_CONT over non-zero gap_scores
      INSERT … ON CONFLICT (stop_id) DO UPDATE  with banded gap_tier.

    Returns rowcount.
    """
    if catchment_radius_m is None:
        catchment_radius_m = _catchment_radius()
    p_high, p_moderate, p_low = _percentile_thresholds()

    result = conn.execute(text("""
        WITH supply AS (
          SELECT s.stop_id,
                 s.agency_id,
                 s.location_type,
                 s.wheelchair_boarding,
                 EXISTS (
                   SELECT 1 FROM gtfs_pathways p
                    WHERE p.agency_id = s.agency_id
                      AND (p.from_stop_id = s.stop_id
                           OR p.to_stop_id = s.stop_id)
                 ) AS has_pathways_data,
                 EXISTS (
                   SELECT 1 FROM gtfs_pathways p
                    WHERE p.agency_id = s.agency_id
                      AND (p.from_stop_id = s.stop_id
                           OR p.to_stop_id = s.stop_id)
                      AND p.pathway_mode <> 2     -- 2 = stairs
                 ) AS has_accessible_pathway
            FROM gtfs_stops s
           WHERE s.geom IS NOT NULL
        ),
        demand AS (
          SELECT s.stop_id,
                 COALESCE(SUM(
                   ca.ambulatory_pop
                   * ST_Area(
                       ST_Intersection(
                         ST_Buffer(s.geom::geography, :r)::geometry,
                         t.geom
                       )
                     )
                   / NULLIF(ST_Area(t.geom), 0)
                 ), 0)::int AS ambulatory_pop_400m
            FROM gtfs_stops s
            LEFT JOIN census_tracts     t  ON ST_DWithin(
                    s.geom::geography, t.geom::geography, :r)
            LEFT JOIN census_ambulatory ca USING (tract_geoid)
           WHERE s.geom IS NOT NULL
           GROUP BY s.stop_id
        ),
        combined AS (
          SELECT supply.stop_id, supply.agency_id,
                 CASE
                   WHEN supply.location_type IN (1, 2)
                    AND supply.has_pathways_data
                    AND NOT supply.has_accessible_pathway
                       THEN 1.0
                   WHEN supply.wheelchair_boarding = 1 THEN 0.0
                   WHEN supply.wheelchair_boarding = 2 THEN 1.0
                   ELSE 0.5
                 END AS supply_penalty,
                 (supply.location_type IN (1, 2)
                    AND supply.has_pathways_data
                    AND NOT supply.has_accessible_pathway) AS pathways_override,
                 demand.ambulatory_pop_400m
            FROM supply
            JOIN demand USING (stop_id)
        ),
        scored AS (
          SELECT *,
                 (supply_penalty * ambulatory_pop_400m)::numeric(10,2)
                   AS gap_score
            FROM combined
        ),
        percentiles AS (
          SELECT PERCENTILE_CONT(:p_high)     WITHIN GROUP (ORDER BY gap_score) AS p_high,
                 PERCENTILE_CONT(:p_moderate) WITHIN GROUP (ORDER BY gap_score) AS p_moderate,
                 PERCENTILE_CONT(:p_low)      WITHIN GROUP (ORDER BY gap_score) AS p_low
            FROM scored
           WHERE gap_score > 0
        )
        INSERT INTO stop_ada_gap (
          stop_id, agency_id, supply_penalty, ambulatory_pop_400m,
          gap_score, gap_tier, pathways_override,
          catchment_radius_m, computed_at
        )
        SELECT scored.stop_id, scored.agency_id,
               scored.supply_penalty, scored.ambulatory_pop_400m,
               scored.gap_score,
               CASE
                 WHEN scored.gap_score <= 0                  THEN 'low'
                 WHEN scored.gap_score >= COALESCE(percentiles.p_high, 0)
                   AND COALESCE(percentiles.p_high, 0) > 0   THEN 'very_high'
                 WHEN scored.gap_score >= COALESCE(percentiles.p_moderate, 0)
                   AND COALESCE(percentiles.p_moderate, 0) > 0 THEN 'high'
                 WHEN scored.gap_score >= COALESCE(percentiles.p_low, 0)
                   AND COALESCE(percentiles.p_low, 0) > 0    THEN 'moderate'
                 ELSE 'low'
               END,
               scored.pathways_override,
               :r,
               now()
          FROM scored, percentiles
        ON CONFLICT (stop_id) DO UPDATE
          SET agency_id            = EXCLUDED.agency_id,
              supply_penalty       = EXCLUDED.supply_penalty,
              ambulatory_pop_400m  = EXCLUDED.ambulatory_pop_400m,
              gap_score            = EXCLUDED.gap_score,
              gap_tier             = EXCLUDED.gap_tier,
              pathways_override    = EXCLUDED.pathways_override,
              catchment_radius_m   = EXCLUDED.catchment_radius_m,
              computed_at          = EXCLUDED.computed_at
    """), {"r": catchment_radius_m,
           "p_high": p_high,
           "p_moderate": p_moderate,
           "p_low": p_low})
    return result.rowcount
```

```python
def get_stop_ada_gap_features(
    conn: Connection,
    *,
    tier: Optional[str] = None,
    limit: int = 5000,
) -> list[dict[str, Any]]:
    """Read stop_ada_gap joined with gtfs_stops geometry + routes
    serving (from gtfs_stop_times + gtfs_trips), return GeoJSON-ready
    Feature dicts.  Sorted by gap_score DESC.
    """
    rows = conn.execute(text("""
        SELECT sag.stop_id,
               sag.agency_id,
               s.stop_name,
               s.wheelchair_boarding,
               sag.supply_penalty,
               sag.ambulatory_pop_400m,
               sag.gap_score,
               sag.gap_tier,
               sag.pathways_override,
               ST_AsGeoJSON(s.geom)::json AS geometry,
               COALESCE(
                 (SELECT array_agg(DISTINCT r.route_short_name
                                   ORDER BY r.route_short_name)
                    FROM gtfs_stop_times st
                    JOIN gtfs_trips      tr ON tr.trip_id  = st.trip_id
                    JOIN gtfs_routes     r  ON r.route_id  = tr.route_id
                                          AND r.agency_id = sag.agency_id
                   WHERE st.stop_id = sag.stop_id), ARRAY[]::text[]
               ) AS routes_serving
          FROM stop_ada_gap sag
          JOIN gtfs_stops    s ON s.stop_id = sag.stop_id
         WHERE (:tier IS NULL OR sag.gap_tier = :tier)
         ORDER BY sag.gap_score DESC NULLS LAST, sag.stop_id
         LIMIT :lim
    """), {"tier": tier, "lim": limit}).mappings().all()

    features: list[dict[str, Any]] = []
    for r in rows:
        features.append({
            "type":     "Feature",
            "geometry": r["geometry"],
            "properties": {
                "stop_id":             r["stop_id"],
                "agency_id":           r["agency_id"],
                "stop_name":           r["stop_name"],
                "wheelchair_boarding": r["wheelchair_boarding"],
                "supply_penalty":      float(r["supply_penalty"])
                                       if r["supply_penalty"] is not None else None,
                "ambulatory_pop_400m": int(r["ambulatory_pop_400m"])
                                       if r["ambulatory_pop_400m"] is not None else 0,
                "gap_score":           float(r["gap_score"])
                                       if r["gap_score"] is not None else 0.0,
                "gap_tier":            r["gap_tier"],
                "pathways_override":   bool(r["pathways_override"]),
                "routes_serving":      list(r["routes_serving"] or []),
            },
        })
    return features
```

```python
def get_stop_ada_gap_breakdown(
    conn: Connection,
    stop_id: str,
) -> dict[str, Any]:
    """Per-stop drill: stop metadata + routes serving + top-3
    contributing tracts by area-weighted ambulatory pop.

    Returns:
      {stop_id, stop_name, agency_id, gap_score, gap_tier,
       supply_penalty, pathways_override, routes: [...], tracts: [...]}
    """
    meta_row = conn.execute(text("""
        SELECT sag.stop_id, sag.agency_id, s.stop_name,
               sag.gap_score, sag.gap_tier, sag.supply_penalty,
               sag.pathways_override, sag.catchment_radius_m
          FROM stop_ada_gap sag
          JOIN gtfs_stops    s ON s.stop_id = sag.stop_id
         WHERE sag.stop_id = :sid
    """), {"sid": stop_id}).mappings().fetchone()

    if not meta_row:
        return {"stop_id": stop_id, "routes": [], "tracts": []}

    routes = conn.execute(text("""
        SELECT r.route_id,
               r.route_short_name,
               COUNT(DISTINCT tr.trip_id)::int AS trips_per_day
          FROM gtfs_stop_times st
          JOIN gtfs_trips      tr ON tr.trip_id  = st.trip_id
          JOIN gtfs_routes     r  ON r.route_id  = tr.route_id
                                AND r.agency_id = :aid
         WHERE st.stop_id = :sid
         GROUP BY r.route_id, r.route_short_name
         ORDER BY trips_per_day DESC
         LIMIT 10
    """), {"sid": stop_id, "aid": meta_row["agency_id"]}).mappings().all()

    tracts = conn.execute(text("""
        WITH stop_geom AS (
          SELECT geom FROM gtfs_stops WHERE stop_id = :sid
        ),
        contributions AS (
          SELECT t.tract_geoid,
                 ca.ambulatory_pop,
                 ST_Area(
                   ST_Intersection(
                     ST_Buffer((SELECT geom FROM stop_geom)::geography, :r)::geometry,
                     t.geom
                   )
                 ) AS overlap_area,
                 ST_Area(t.geom) AS tract_area
            FROM census_tracts     t
            JOIN census_ambulatory ca USING (tract_geoid)
           WHERE ST_DWithin(
                  (SELECT geom FROM stop_geom)::geography,
                  t.geom::geography, :r)
        )
        SELECT tract_geoid,
               ambulatory_pop,
               ROUND(overlap_area / NULLIF(tract_area, 0), 3)::float
                 AS area_weighted_share
          FROM contributions
         ORDER BY (ambulatory_pop * overlap_area / NULLIF(tract_area, 0)) DESC
         LIMIT 3
    """), {"sid": stop_id,
           "r": meta_row["catchment_radius_m"] or _DEFAULT_CATCHMENT_M}
         ).mappings().all()

    return {
        "stop_id":           meta_row["stop_id"],
        "stop_name":         meta_row["stop_name"],
        "agency_id":         meta_row["agency_id"],
        "gap_score":         float(meta_row["gap_score"]) if meta_row["gap_score"] is not None else None,
        "gap_tier":          meta_row["gap_tier"],
        "supply_penalty":    float(meta_row["supply_penalty"]) if meta_row["supply_penalty"] is not None else None,
        "pathways_override": bool(meta_row["pathways_override"]),
        "routes":  [{"route_id":         r["route_id"],
                     "route_short_name": r["route_short_name"],
                     "trips_per_day":    int(r["trips_per_day"])}
                    for r in routes],
        "tracts":  [{"tract_geoid":         t["tract_geoid"],
                     "ambulatory_pop":      int(t["ambulatory_pop"] or 0),
                     "area_weighted_share": float(t["area_weighted_share"] or 0.0)}
                    for t in tracts],
    }
```

```tsx
  stop_name: string;
  wheelchair_boarding: number | null;
  supply_penalty: number | null;
  ambulatory_pop_400m: number;
  gap_score: number;
  gap_tier: "low" | "moderate" | "high" | "very_high" | null;
  pathways_override: boolean;
  routes_serving: string[];

  type: "Feature";
  geometry: { type: "Point"; coordinates: [number, number] };

export function ADAGap(): JSX.Element {

  const [loading, setLoading] = useState(true);
  const [error, setError]     = useState<string | null>(null);

  useEffect(() => {
    let cancelled = false;
    setLoading(true);
    setError(null);
    api.get<AdaStopFC>("/accessibility/gaps", { params: { limit: 20000 } })
       .then(r => { if (!cancelled) setFc(r.data); })
       .catch(e => { if (!cancelled) setError(e.message ?? "fetch failed"); })
       .finally(() => { if (!cancelled) setLoading(false); });
    return () => { cancelled = true; };
  }, []);

  if (error) return <SectionError message={error} />;

  const features = fc?.features ?? [];
  const total       = features.length;
  const inaccessible = features.filter(
    f => f.properties.wheelchair_boarding === 2).length;
  const unknown      = features.filter(
    f => f.properties.wheelchair_boarding === 0
      || f.properties.wheelchair_boarding === null).length;
  const overrides    = features.filter(
    f => f.properties.pathways_override).length;
  const totalUnderserved = features
    .filter(f => f.properties.gap_tier === "very_high")
    .reduce((acc, f) => acc + (f.properties.gap_score ?? 0), 0);

  const top10 = features.slice(0, 10);

      f.properties.agency_id,
      JSON.stringify(f.properties.stop_name ?? ""),
      String(f.properties.gap_score ?? 0),
      f.properties.gap_tier ?? "",
      String(f.properties.ambulatory_pop_400m ?? 0),
      String(f.properties.supply_penalty ?? 0),
      String(f.properties.wheelchair_boarding ?? ""),
      String(f.properties.pathways_override),
    ].join(","));
    return [header.join(","), ...rows].join("\n");

  return (
    <section className="section section--ada-gap">
      <header className="section__header">
        <h2>ADA Accessibility Gap</h2>

        </p>

        <KpiCard label="Inaccessible stops"     value={inaccessible} loading={loading} />
        <KpiCard label="Unknown-status stops"   value={unknown}      loading={loading} />
        <KpiCard label="Pathways-overridden"    value={overrides}    loading={loading} />
        <KpiCard label="Total underserved"      value={Math.round(totalUnderserved).toLocaleString()}
                                                loading={loading} />
      </div>

      <SectionMap mapMode="ada-gap" />

      <div className="section__panel">
        <header className="section__panel-header">
          <h3>Top 10 stops to retrofit</h3>
          <ExportButton onExport={exportCsv}
                        filename="ada-gap-stops.csv" />
        </header>
        <table className="data-table">
          <thead>
            <tr>
              <th>Stop</th><th>Agency</th><th>Gap score</th>
              <th>Tier</th><th>Ambulatory pop</th><th>Supply penalty</th>
```

```python
#!/usr/bin/env python3
"""Sprint 5c acceptance script — runs 11 live checks against a
running BATIS backend and dev DB.  Returns exit 0 on full pass.

Mirrors scripts/sprint-5b-acceptance.py.
"""
from __future__ import annotations

import os
import sys
import time
from typing import Any

import requests
from sqlalchemy import create_engine, text


BACKEND = os.environ.get("BATIS_BACKEND_URL", "http://localhost:8001")
FRONTEND = os.environ.get("BATIS_FRONTEND_URL", "http://localhost:5173")
API_KEY  = os.environ.get("BACKEND_API_KEY", "")
DB_URL   = os.environ.get(
    "DATABASE_URL",
    "postgresql://postgres:password@localhost:5432/bay_area_transit_intel"
)

_engine = create_engine(DB_URL)
_HEADERS = {"X-API-Key": API_KEY} if API_KEY else {}


def _ok(label: str, msg: str = "") -> None:
    print(f"  \033[32mPASS\033[0m  {label}  {msg}".rstrip())


def _soft(label: str, msg: str) -> None:
    print(f"  \033[33mPASS\033[0m  {label}  (note: {msg})")


def _fail(label: str, msg: str) -> None:
    print(f"  \033[31mFAIL\033[0m  {label}  {msg}")


_RESULTS: list[tuple[str, bool]] = []


def check(label: str) -> Any:
    def decorator(fn):
        try:
            fn()
            _RESULTS.append((label, True))
        except SoftPass as e:
            _soft(label, str(e))
            _RESULTS.append((label, True))
        except Exception as e:  # noqa: BLE001
            _fail(label, str(e))
            _RESULTS.append((label, False))
        return fn
    return decorator


class SoftPass(Exception):
    pass


# ── checks ───────────────────────────────────────────────────────────────────

@check("1. Migration applied")
def _1():
    with _engine.connect() as conn:
        tables = {r[0] for r in conn.execute(text(
          "SELECT table_name FROM information_schema.tables "
          "WHERE table_schema = 'public'"
        ))}
    expected = {"census_tracts","gtfs_pathways","gtfs_levels",
                "census_ambulatory","stop_ada_gap"}
    missing = expected - tables
    if missing:
        raise RuntimeError(f"missing tables: {missing}")
    _ok("1. Migration applied", f"({len(expected)} tables present)")


@check("2. Tract geometry loaded")
def _2():
    with _engine.connect() as conn:
        n = conn.execute(text("SELECT COUNT(*) FROM census_tracts")).scalar()
    if n < 1500:
        raise RuntimeError(f"only {n} tracts (expected ≥ 1500)")
    _ok("2. Tract geometry loaded", f"({n} tracts)")


@check("3. B18105 loaded")
def _3():
    with _engine.connect() as conn:
        n = conn.execute(text("SELECT COUNT(*) FROM census_ambulatory")).scalar()
    if n < 1500:
        raise RuntimeError(f"only {n} tracts in census_ambulatory")
    _ok("3. B18105 loaded", f"({n} tracts)")


@check("4. Pathways loaded for ≥1 rail agency")
def _4():
    with _engine.connect() as conn:
        n = conn.execute(text("SELECT COUNT(*) FROM gtfs_pathways")).scalar()
    if n == 0:
        raise SoftPass("0 pathway rows — most agencies don't publish "
                       "pathways.txt; this is expected if no rail agency "
                       "feed was downloaded yet")
    _ok("4. Pathways loaded for ≥1 rail agency", f"({n} rows)")


@check("5. stop_ada_gap populated")
def _5():
    with _engine.connect() as conn:
        row = conn.execute(text(
          "SELECT COUNT(*), COUNT(*) FILTER (WHERE gap_score IS NULL) "
          "FROM stop_ada_gap"
        )).first()
        n, null_cnt = int(row[0]), int(row[1])
    if n < 1000:
        raise RuntimeError(f"only {n} stops scored (expected ≥ 1000)")
    if null_cnt > 0:
        raise RuntimeError(f"{null_cnt} rows have null gap_score")
    _ok("5. stop_ada_gap populated", f"({n} stops, 0 null gap_score)")


@check("6. GeoJSON shape")
def _6():
    r = requests.get(f"{BACKEND}/accessibility/gaps",
                     params={"limit": 50}, headers=_HEADERS, timeout=30)
    r.raise_for_status()
    body = r.json()
    if body.get("type") != "FeatureCollection":
        raise RuntimeError(f"unexpected type: {body.get('type')}")
    feats = body.get("features") or []
    if not feats:
        raise RuntimeError("empty features array")
    expected_props = {"stop_id","agency_id","stop_name",
                      "wheelchair_boarding","supply_penalty",
                      "ambulatory_pop_400m","gap_score","gap_tier",
                      "pathways_override","routes_serving"}
    missing = expected_props - set(feats[0]["properties"].keys())
    if missing:
        raise RuntimeError(f"missing properties: {missing}")
    scores = [f["properties"]["gap_score"] for f in feats]
    if scores != sorted(scores, reverse=True):
        raise RuntimeError("features not sorted by gap_score DESC")
    _ok("6. GeoJSON shape", f"({len(feats)} features, all props present)")


@check("7. Tier distribution sensible")
def _7():
    with _engine.connect() as conn:
        rows = conn.execute(text(
          "SELECT gap_tier, COUNT(*) FROM stop_ada_gap GROUP BY gap_tier"
        )).all()
    tiers = {r[0]: r[1] for r in rows}
    total = sum(tiers.values())
    for needed in ("low","moderate","high","very_high"):
        if tiers.get(needed, 0) < 1:
            raise RuntimeError(f"tier '{needed}' has 0 stops")
    if tiers.get("very_high", 0) > 0.15 * total:
        raise RuntimeError(
          f"very_high = {tiers['very_high']} > 15% of {total} — "
          "percentile banding may be miscomputed"
        )
    _ok("7. Tier distribution sensible",
        f"low={tiers.get('low',0)} mod={tiers.get('moderate',0)} "
        f"high={tiers.get('high',0)} vh={tiers.get('very_high',0)}")
```
